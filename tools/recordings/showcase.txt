# The 30 s showcase video (tools/make-media.ps1 renders it at 1920x1080 and encodes it).
fps 60
duration 30

# 0-6 s, day: PBR materials and soft sun shadows, through the arch and down the hall.
set 0 points.count 0
set 0 sun.azimuth -120
set 0 sun.elevation 78
set 0 tonemap.exposure 0.5
camera 0    -11.5 1.7  0.3   90   4
camera 3     -4.0 2.0  0.3   92   6
camera 6     -1.5 2.4  0.3   93   6
# 6-11 s: the sun sweeps across the sky and the shadows follow, while the camera rises and looks down.
ramp 6 11 sun.azimuth -120 -20
ramp 6 11 sun.elevation 78 72
camera 11     1.5 3.6  0.3   90 -12
# 11-20 s: night falls and 512 animated lamps come up, the nearest 32 with cube shadow maps.
set 11 points.shadows cube
set 11 points.budget 32
ramp 11 13.5 sun.intensity 8 0
ramp 11 13.5 ambient.intensity 0.25 0.02
ramp 11 13.5 tonemap.exposure 0.5 0.0
ramp 11.5 14.5 points.count 0 512
camera 13     1.0 2.4  0.3  180  -2
camera 15     0.0 2.0  0.3  265  -4
camera 17    -3.5 1.9  0.3  268  -2
camera 20    -7.5 2.1  0.3  450   0
# 20-25 s: what the shaders see.
set 20 view.show basecolor
set 21.25 view.show normals
set 22.5 view.show clusterlights
set 23.75 view.show pointshadowdiff
camera 25    -3.0 2.0  0.3  452   2
# 25-30 s: cut to the side arcade (the title card goes on top).
set 25 view.show lit
camera 25     5.0 1.3  4.3  248  -6
camera 30     2.0 1.45 4.4  252  -3
